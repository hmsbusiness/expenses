Apps Script Web App URL
This single URL handles everything: Sheet, Drive, and email. See the Setup Guide.
How it works

Everything routes through a single Google Apps Script web app — no OAuth login required. When someone submits a report, the form sends all data and files to Apps Script, which writes to Google Sheets, uploads receipts to Drive, and sends the email notification. The submitter gets a downloaded PDF summary.

Step 1 — Set up your Google Sheet
1

Create a Google Sheet with a tab named Expenses. Row 1 headers:

Timestamp | Name | Email | Classroom/Admin | Report Title | Period | Merchant | Description | Category | Date | Miles | Amount | Notes | Receipts
2

Copy the Sheet ID from its URL: docs.google.com/spreadsheets/d/SHEET-ID-HERE/edit

Step 2 — Create a Drive folder for receipts
1

Create a folder in Google Drive for receipt uploads.

2

Copy the folder ID from its URL: drive.google.com/drive/folders/FOLDER-ID-HERE

Step 3 — Create the Apps Script
1

Go to script.google.com → New project.

2

Replace all existing code with the script below. Fill in your SHEET_ID and FOLDER_ID at the top.

// ── CONFIGURATION — fill these in ───────────────────────── var SHEET_ID = "YOUR_SHEET_ID_HERE"; var FOLDER_ID = "YOUR_DRIVE_FOLDER_ID_HERE"; var NOTIFY = ["office@hollismontessori.org", "sshor@hollismontessori.org"]; // ────────────────────────────────────────────────────────── function doPost(e) { try { var data = JSON.parse(e.postData.contents); // 1. Upload receipt files to Drive var driveLinks = {}; if (data.files && data.files.length) { var folder = DriveApp.getFolderById(FOLDER_ID); data.files.forEach(function(f) { var bytes = Utilities.base64Decode(f.base64); var blob = Utilities.newBlob(bytes, f.mimeType, f.name); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); driveLinks[f.name] = file.getUrl(); }); } // 2. Write rows to Google Sheet var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheetByName("Expenses"); var ts = new Date().toISOString(); var base = [ts, data.name, data.email, data.dept, data.title, data.period]; var allLinks = Object.values(driveLinks).join(", "); var rows = []; (data.expenseItems || []).forEach(function(r) { rows.push(base.concat([r.merchant, r.desc, r.cat, r.date, "", r.amount, data.notes, allLinks])); }); (data.mileageItems || []).forEach(function(r) { rows.push(base.concat([r.merchant, r.desc, "Mileage", r.date, r.miles, r.amount, data.notes, allLinks])); }); if (rows.length) { sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, rows[0].length) .setValues(rows); } // 3. Send email notification var subject = "New Expense Report: " + (data.title || "Untitled") + " — " + (data.name || "Unknown"); var body = "A new expense report has been submitted.\n\n" + "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n" + "Submitted by: " + (data.name || "—") + "\n" + "Email: " + (data.email || "—") + "\n" + "Classroom/Admin: " + (data.dept || "—") + "\n" + "Report title: " + (data.title || "—") + "\n" + "Period: " + (data.period || "—") + "\n" + "Submitted: " + (data.submittedAt || "—") + "\n" + "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n\n"; if (data.expenseItems && data.expenseItems.length) { body += "EXPENSES\n"; data.expenseItems.forEach(function(r) { body += " • " + (r.merchant||"—") + " | " + (r.desc||"—") + " | " + r.cat + " | " + r.date + " | $" + parseFloat(r.amount||0).toFixed(2) + "\n"; }); body += "\n"; } if (data.mileageItems && data.mileageItems.length) { body += "MILEAGE\n"; data.mileageItems.forEach(function(r) { body += " • " + (r.desc||"—") + " | " + (r.merchant||"—") + " | " + r.date + " | " + r.miles + " mi" + " | $" + parseFloat(r.amount||0).toFixed(2) + "\n"; }); body += "\n"; } body += "TOTAL: $" + parseFloat(data.total||0).toFixed(2) + "\n\n"; if (data.notes) body += "NOTES\n" + data.notes + "\n\n"; if (Object.keys(driveLinks).length) { body += "RECEIPT LINKS\n"; Object.entries(driveLinks).forEach(function(kv) { body += " • " + kv[0] + ": " + kv[1] + "\n"; }); } body += "\n— Hollis Montessori Expense Report System"; MailApp.sendEmail({ to: NOTIFY.join(","), subject: subject, body: body }); return ContentService .createTextOutput(JSON.stringify({ status: "ok", driveLinks: driveLinks })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService .createTextOutput(JSON.stringify({ status: "ok" })) .setMimeType(ContentService.MimeType.JSON); }
3

Fill in SHEET_ID and FOLDER_ID at the top, then click Save.

4

Click Deploy → New deployment. Type: Web app. Execute as: Me. Who has access: Anyone. Click Deploy and authorize all permissions.

5

Copy the Web app URL (ends in /exec) and paste it into the Configuration tab, then click Save Configuration.

Re-deploying after changes

If you update the Apps Script code, use Deploy → Manage deployments → Edit → New version to update in place. The URL stays the same.
